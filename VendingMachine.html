    USERS {
        int user_id PK
        varchar name
        varchar email
        varchar phone
        datetime created_at
    }

    LOCATIONS {
        int location_id PK
        varchar building_name
        varchar address
        varchar city
        varchar state
        varchar pincode
    }

    VENDING_MACHINES {
        int machine_id PK
        int location_id FK
        varchar machine_code
        varchar status
        datetime installed_at
    }

    PRODUCT_CATEGORIES {
        int category_id PK
        varchar category_name
        varchar description
    }

    PRODUCTS {
        int product_id PK
        int category_id FK
        varchar product_name
        varchar brand
        decimal default_price
        int expiry_days
        varchar status
    }

    MACHINE_SLOTS {
        int slot_id PK
        int machine_id FK
        int product_id FK
        varchar slot_code
        int capacity
        int current_quantity
    }

    INVENTORY {
        int inventory_id PK
        int machine_id FK
        int product_id FK
        int quantity
        int minimum_quantity
        datetime updated_at
    }

    PRICES {
        int price_id PK
        int product_id FK
        decimal price
        date start_date
        date end_date
    }

    ORDERS {
        int order_id PK
        int user_id FK
        int machine_id FK
        decimal total_amount
        varchar order_status
        datetime created_at
    }

    ORDER_ITEMS {
        int order_item_id PK
        int order_id FK
        int product_id FK
        int quantity
        decimal unit_price
        decimal subtotal
    }

    PAYMENTS {
        int payment_id PK
        int order_id FK
        decimal amount
        varchar payment_method
        varchar transaction_id
        varchar payment_status
        datetime paid_at
    }

    REFUNDS {
        int refund_id PK
        int payment_id FK
        decimal amount
        varchar reason
        varchar refund_status
        datetime refunded_at
    }

    TRANSACTIONS {
        int transaction_id PK
        int machine_id FK
        varchar transaction_type
        decimal amount
        datetime created_at
    }

    MACHINE_MAINTENANCE {
        int maintenance_id PK
        int machine_id FK
        varchar maintenance_type
        text description
        varchar status
        datetime maintenance_date
    }

    MACHINE_ALERTS {
        int alert_id PK
        int machine_id FK
        varchar alert_type
        varchar message
        varchar severity
        boolean resolved
        datetime created_at
    }
