Hotel Booking Database ER Diagram

erDiagram

    USERS ||--o{ USER_ADDRESSES : has
    USERS ||--o{ BOOKINGS : makes
    USERS ||--o{ REVIEWS : writes

    HOTELS ||--|| HOTEL_ADDRESSES : has
    HOTELS ||--o{ HOTEL_IMAGES : contains

    HOTELS ||--o{ ROOMS : contains
    ROOM_TYPES ||--o{ ROOMS : categorizes
    ROOMS ||--o{ ROOM_IMAGES : contains

    HOTELS ||--o{ HOTEL_AMENITIES : provides
    AMENITIES ||--o{ HOTEL_AMENITIES : included_in

    ROOMS ||--o{ ROOM_AMENITIES : provides
    AMENITIES ||--o{ ROOM_AMENITIES : included_in

    BOOKINGS ||--o{ BOOKING_ROOMS : contains
    ROOMS ||--o{ BOOKING_ROOMS : booked_as

    BOOKINGS ||--o{ GUESTS : includes
    BOOKINGS ||--o{ PAYMENTS : has

    BOOKINGS ||--o{ REFUNDS : generates
    COUPONS ||--o{ BOOKINGS : applied_to

    HOTELS ||--o{ REVIEWS : receives


    USERS {
        int user_id PK
        varchar name
        varchar email
        varchar phone
        varchar password
        datetime created_at
    }

    USER_ADDRESSES {
        int address_id PK
        int user_id FK
        varchar address
        varchar city
        varchar state
        varchar country
        varchar pincode
    }

    HOTELS {
        int hotel_id PK
        varchar hotel_name
        text description
        decimal rating
        varchar contact_number
        varchar email
        varchar status
    }

    HOTEL_ADDRESSES {
        int hotel_address_id PK
        int hotel_id FK
        varchar address
        varchar city
        varchar state
        varchar country
        varchar pincode
        decimal latitude
        decimal longitude
    }

    HOTEL_IMAGES {
        int image_id PK
        int hotel_id FK
        varchar image_url
    }

    ROOM_TYPES {
        int room_type_id PK
        varchar type_name
        text description
        int max_guests
        decimal base_price
    }

    ROOMS {
        int room_id PK
        int hotel_id FK
        int room_type_id FK
        varchar room_number
        varchar floor
        varchar status
    }

    ROOM_IMAGES {
        int image_id PK
        int room_id FK
        varchar image_url
    }

    AMENITIES {
        int amenity_id PK
        varchar amenity_name
        varchar description
    }

    HOTEL_AMENITIES {
        int hotel_id PK
        int amenity_id FK
    }

    ROOM_AMENITIES {
        int room_id PK
        int amenity_id FK
    }

    BOOKINGS {
        int booking_id PK
        int user_id FK
        int coupon_id FK
        date check_in
        date check_out
        int guests_count
        decimal total_amount
        varchar booking_status
        datetime created_at
    }

    BOOKING_ROOMS {
        int booking_room_id PK
        int booking_id FK
        int room_id FK
        decimal price
    }

    GUESTS {
        int guest_id PK
        int booking_id FK
        varchar name
        int age
        varchar gender
        varchar id_proof
    }

    PAYMENTS {
        int payment_id PK
        int booking_id FK
        decimal amount
        varchar payment_method
        varchar transaction_id
        varchar payment_status
        datetime paid_at
    }

    COUPONS {
        int coupon_id PK
        varchar coupon_code
        decimal discount_percentage
        decimal max_discount
        date valid_from
        date valid_until
        varchar status
    }

    REVIEWS {
        int review_id PK
        int user_id FK
        int hotel_id FK
        int rating
        text comment
        datetime created_at
    }

    REFUNDS {
        int refund_id PK
        int payment_id FK
        decimal refund_amount
        varchar reason
        varchar refund_status
        datetime refunded_at
    }
