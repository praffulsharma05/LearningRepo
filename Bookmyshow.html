ANALYSIS VERDICT: SUBSTANTIAL EXTENSIONS IDENTIFIED ON LIVE WEBSITE
Comparing Provided Image ER Diagram vs Live in.bookmyshow.com Platform
The image provides a strong academic / core database schema for event and movie ticketing. However, analyzing the live website https://in.bookmyshow.com/ reveals several critical real-world architectural requirements missing in the image diagram.

Provided Image ER Diagram Features

  • Core Entities Covered: Users, Movies, Shows, Theatres, Screens, Seats, Show_Seats, Bookings, Payments.
  • Cast & Crew Mapping: People, Cast/Crew roles, Director & Actor associations.
  • Event Module: Basic Events, Event Venues, Performers, Categories.
  • F&B Add-on: Meals / Food Items linked to Bookings.
  • Offers & Discounts: Offers, Coupons, Gift Cards, Wallet transactions.
  • Basic Stream Content: Stream Movies, Buy/Rent pricing models.

Insights Discovered from Live bookmyshow.com URL

  • Multi-Format & Multi-Language Matrix: A single movie has distinct showtimes based on Format (2D, 3D, IMAX 3D, 4DX, ScreenX, ICE, Dolby Atmos) AND Language (Hindi, English, Telugu, Tamil, Kannada).
  • Geographic Location Scoping: Shows are contextual to selected City / Location (e.g. Mumbai, Bengaluru, NCR) with city-specific theatre availability.
  • Transient Seat Locking (Concurrency Engine): When selecting seats, BookMyShow locks them for 10 minutes (TTL countdown) using Redis/DB locks with locked_at and locked_by_user_id.
  • Cancellation Protect Policy: Optional add-on insurance per booking enabling 75%-100% refund upon cancellation before deadline.
  • F&B Seat Delivery vs Pickup Counter: Food can be delivered directly to seat during interval or kept ready at express pickup counter.
  • Ticket Types for Live Events: Events do NOT use screen seat grids; they use tiered inventory passes (Early Bird, Phase 1, Phase 2, VIP, Fan Pit) with stock counts.
  • M-Ticket (QR Code) & Gate Scanner Usher App: Digital QR token with validation state tracking for entry ushers.

Production Relational Database DDL Schema (PostgreSQL Dialect)
Includes foreign key constraints, high-concurrency seat locking indexes, and ENUM types for BookMyShow backend deployment. 
-- 1. CITIES TABLE
CREATE TABLE cities (
    city_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    state VARCHAR(100) NOT NULL,
    is_active BOOLEAN DEFAULT TRUE
);

-- 2. USERS TABLE
CREATE TABLE users (
    user_id SERIAL PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    email VARCHAR(150) UNIQUE NOT NULL,
    phone VARCHAR(20) UNIQUE NOT NULL,
    city_id INT REFERENCES cities(city_id),
    is_superstar BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 3. MOVIES TABLE
CREATE TABLE movies (
    movie_id SERIAL PRIMARY KEY,
    title VARCHAR(200) NOT NULL,
    duration_mins INT NOT NULL,
    censor_rating VARCHAR(10) CHECK (censor_rating IN ('U', 'UA', 'A', 'S')),
    release_date DATE NOT NULL,
    user_rating_pct DECIMAL(5,2) DEFAULT 0.00
);

-- 4. THEATRES & SCREENS
CREATE TABLE theatres (
    theatre_id SERIAL PRIMARY KEY,
    city_id INT REFERENCES cities(city_id) ON DELETE CASCADE,
    name VARCHAR(150) NOT NULL,
    address TEXT NOT NULL,
    cancellation_supported BOOLEAN DEFAULT TRUE
);

CREATE TABLE screens (
    screen_id SERIAL PRIMARY KEY,
    theatre_id INT REFERENCES theatres(theatre_id) ON DELETE CASCADE,
    screen_name VARCHAR(50) NOT NULL,
    total_seats INT NOT NULL
);

-- 5. SEATS & SHOWS
CREATE TABLE seat_categories (
    category_id SERIAL PRIMARY KEY,
    name VARCHAR(50) NOT NULL -- Recliner, Club, Executive
);

CREATE TABLE seats (
    seat_id SERIAL PRIMARY KEY,
    screen_id INT REFERENCES screens(screen_id) ON DELETE CASCADE,
    category_id INT REFERENCES seat_categories(category_id),
    row_name VARCHAR(5) NOT NULL,
    seat_number INT NOT NULL,
    UNIQUE(screen_id, row_name, seat_number)
);

CREATE TABLE shows (
    show_id SERIAL PRIMARY KEY,
    movie_id INT REFERENCES movies(movie_id),
    screen_id INT REFERENCES screens(screen_id),
    start_time TIMESTAMP NOT NULL,
    end_time TIMESTAMP NOT NULL,
    format VARCHAR(20) NOT NULL, -- 2D, 3D, IMAX 3D, 4DX
    language VARCHAR(50) NOT NULL
);

-- 6. SHOW SEATS & CONCURRENCY SEAT LOCKING
TYPE SEAT_STATUS AS ENUM ('AVAILABLE', 'LOCKED', 'BOOKED');

CREATE TABLE show_seats (
    show_seat_id BIGSERIAL PRIMARY KEY,
    show_id INT REFERENCES shows(show_id) ON DELETE CASCADE,
    seat_id INT REFERENCES seats(seat_id),
    status SEAT_STATUS DEFAULT 'AVAILABLE',
    lock_expires_at TIMESTAMP NULL,
    locked_by_user_id INT REFERENCES users(user_id) NULL,
    UNIQUE(show_id, seat_id)
);

-- High Performance Index for Real-time Seat Lock Lookup
CREATE INDEX idx_show_seat_concurrency ON show_seats(show_id, status, lock_expires_at);

-- 7. BOOKINGS & PAYMENTS
CREATE TABLE bookings (
    booking_id SERIAL PRIMARY KEY,
    booking_ref_no VARCHAR(30) UNIQUE NOT NULL,
    user_id INT REFERENCES users(user_id),
    show_id INT REFERENCES shows(show_id),
    total_amount DECIMAL(10,2) NOT NULL,
    discount_amount DECIMAL(10,2) DEFAULT 0.00,
    net_amount DECIMAL(10,2) NOT NULL,
    status VARCHAR(30) DEFAULT 'CONFIRMED',
    qr_code_token VARCHAR(255) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE payments (
    payment_id SERIAL PRIMARY KEY,
    booking_id INT UNIQUE REFERENCES bookings(booking_id),
    transaction_ref VARCHAR(100) UNIQUE NOT NULL,
    payment_mode VARCHAR(50) NOT NULL, -- UPI, CREDIT_CARD, NET_BANKING, BMS_WALLET
    amount DECIMAL(10,2) NOT NULL,
    status VARCHAR(30) DEFAULT 'SUCCESS',
    paid_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
